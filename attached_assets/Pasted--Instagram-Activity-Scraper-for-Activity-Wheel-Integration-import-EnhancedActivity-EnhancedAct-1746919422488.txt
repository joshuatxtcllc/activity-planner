// Instagram Activity Scraper for Activity Wheel Integration

import { EnhancedActivity, EnhancedActivityWheel } from './activity-wheel-enhancements';

class InstagramActivityScraper {
  constructor(activityWheel) {
    this.activityWheel = activityWheel;
    this.savedPosts = [];
    this.influencerPosts = [];
    this.processingQueue = [];
    
    // Keywords to identify different activity types
    this.categoryKeywords = {
      'FOOD_DRINK': [
        'restaurant', 'cafe', 'coffee', 'brunch', 'dinner', 'lunch', 'food', 'drinks',
        'bar', 'brewery', 'wine', 'cocktail', 'eating', 'foodie', 'meal'
      ],
      'OUTDOOR': [
        'hike', 'trail', 'park', 'outdoor', 'nature', 'beach', 'lake', 'kayak',
        'bike', 'walk', 'garden', 'picnic', 'camping', 'adventure'
      ],
      'ENTERTAINMENT': [
        'show', 'concert', 'movie', 'theater', 'performance', 'music', 'comedy',
        'festival', 'event', 'nightlife', 'club', 'party'
      ],
      'CULTURE': [
        'museum', 'gallery', 'art', 'exhibition', 'historic', 'tour', 'cultural',
        'workshop', 'class', 'learn', 'experience'
      ],
      'ACTIVE': [
        'gym', 'fitness', 'yoga', 'pilates', 'sports', 'workout', 'exercise',
        'run', 'marathon', 'competition', 'active'
      ],
      'RELAXING': [
        'spa', 'massage', 'wellness', 'relax', 'meditation', 'bath', 'sauna',
        'retreat', 'quiet', 'peaceful'
      ],
      'SHOPPING': [
        'shop', 'store', 'boutique', 'market', 'mall', 'retail', 'buy'
      ]
    };
    
    // Common cost indicators
    this.costKeywords = {
      'FREE': ['free', 'no charge', 'complimentary', '$0'],
      'LOW': ['cheap', 'budget', 'affordable', '$', 'under 10', 'under 15'],
      'MEDIUM': ['moderate', '$$', '15-30', '20-40', 'reasonable'],
      'HIGH': ['expensive', 'luxury', '$$$', 'premium', 'high-end', 'upscale']
    };
    
    // Time commitment indicators
    this.timeKeywords = {
      'QUICK': ['quick', 'brief', '15 min', '30 min', 'short', 'express'],
      'SHORT': ['1 hour', '2 hour', 'couple hours', 'afternoon'],
      'MEDIUM': ['half day', 'morning', 'evening', '3-4 hour'],
      'LONG': ['all day', 'full day', 'day trip', '8 hour'],
      'MULTI_DAY': ['weekend', 'multiple days', 'overnight', 'stay']
    };
  }
  
  // Main method to process Instagram posts
  async scrapeInstagramPosts(posts, options = {}) {
    const {
      autoAddToWheel = false,
      requireManualApproval = true,
      filterByLocation = true,
      locationRadius = 50, // miles
      userLocation = null
    } = options;
    
    for (const post of posts) {
      try {
        const activityData = await this.extractActivityFromPost(post);
        
        if (activityData) {
          if (filterByLocation && userLocation) {
            // Check if location is within radius (you'd need geolocation API)
            const distance = this.calculateDistance(userLocation, activityData.coordinates);
            if (distance > locationRadius) {
              continue; // Skip activities too far away
            }
          }
          
          // Create enhanced activity
          const activity = new EnhancedActivity({
            name: activityData.name,
            description: activityData.description,
            category: activityData.category,
            costLevel: activityData.costLevel,
            timeCommitment: activityData.timeCommitment,
            location: activityData.location,
            isUserAdded: false, // Mark as scraped from Instagram
            url: post.url,
            imageUrl: activityData.imageUrl,
            tags: activityData.tags,
            seasonality: activityData.seasonality
          });
          
          if (autoAddToWheel && !requireManualApproval) {
            this.activityWheel.addActivity(activity);
          } else {
            // Add to queue for manual approval
            this.processingQueue.push({
              activity: activity,
              post: post,
              confidence: activityData.confidence
            });
          }
        }
      } catch (error) {
        console.error(`Error processing post ${post.id}:`, error);
      }
    }
    
    return this.processingQueue;
  }
  
  // Extract activity data from a single Instagram post
  async extractActivityFromPost(post) {
    try {
      // Get text content (caption + comments)
      const text = `${post.caption || ''} ${(post.comments || []).join(' ')}`.toLowerCase();
      
      // Extract location information
      const location = this.extractLocation(post, text);
      if (!location.name) return null; // Skip posts without clear location
      
      // Determine activity name
      const name = this.extractActivityName(text, location.name);
      
      // Extract description
      const description = this.extractDescription(text);
      
      // Categorize the activity
      const category = this.categorizeActivity(text);
      
      // Estimate cost level
      const costLevel = this.estimateCost(text);
      
      // Estimate time commitment
      const timeCommitment = this.estimateTime(text);
      
      // Extract hashtags as tags
      const tags = this.extractTags(post);
      
      // Check seasonality
      const seasonality = this.checkSeasonality(text, post.timestamp);
      
      // Calculate confidence score
      const confidence = this.calculateConfidence({
        hasLocation: !!location.name,
        hasCategory: !!category,
        hasDescription: description.length > 10,
        hasTags: tags.length > 0
      });
      
      return {
        name: name,
        description: description,
        category: category,
        costLevel: costLevel,
        timeCommitment: timeCommitment,
        location: location.name,
        coordinates: location.coordinates,
        tags: tags,
        seasonality: seasonality,
        imageUrl: post.imageUrl || post.images?.[0]?.url,
        confidence: confidence
      };
    } catch (error) {
      console.error('Error extracting activity from post:', error);
      return null;
    }
  }
  
  // Extract location from post
  extractLocation(post, text) {
    // Check if post has location tag
    if (post.location) {
      return {
        name: post.location.name,
        coordinates: post.location.coordinates
      };
    }
    
    // Look for location indicators in text
    const locationPatterns = [
      /at\s+([^.!?,]+[a-zA-Z])/,
      /visiting\s+([^.!?,]+[a-zA-Z])/,
      /check out\s+([^.!?,]+[a-zA-Z])/,
      /location:\s*([^.!?,]+[a-zA-Z])/,
      /@\s*([^.!?,\s]+\s[^.!?,\s]+)/
    ];
    
    for (const pattern of locationPatterns) {
      const match = text.match(pattern);
      if (match) {
        return {
          name: match[1].trim(),
          coordinates: null // Would need geocoding API
        };
      }
    }
    
    return { name: null, coordinates: null };
  }
  
  // Extract activity name
  extractActivityName(text, locationName) {
    // Try to find specific activity mentions
    const activityPatterns = [
      /try\s+the\s+([^.!?,]+)/,
      /ordered\s+the\s+([^.!?,]+)/,
      /get\s+the\s+([^.!?,]+)/,
      /must\s+try\s+([^.!?,]+)/
    ];
    
    for (const pattern of activityPatterns) {
      const match = text.match(pattern);
      if (match) {
        return `${match[1].trim()} at ${locationName}`;
      }
    }
    
    // Default to location name
    return locationName;
  }
  
  // Extract description
  extractDescription(text) {
    // Remove mentions, hashtags, and URLs
    let cleanText = text
      .replace(/@\w+/g, '')
      .replace(/#\w+/g, '')
      .replace(/https?:\/\/\S+/g, '')
      .trim();
    
    // Take first few sentences
    const sentences = cleanText.split(/[.!?]+/);
    return sentences.slice(0, 2).join('. ').trim() + '.';
  }
  
  // Categorize activity based on keywords
  categorizeActivity(text) {
    let scores = {};
    
    for (const [category, keywords] of Object.entries(this.categoryKeywords)) {
      scores[category] = 0;
      
      for (const keyword of keywords) {
        if (text.includes(keyword)) {
          scores[category]++;
        }
      }
    }
    
    // Return category with highest score
    const topCategory = Object.entries(scores).reduce((a, b) => a[1] > b[1] ? a : b)[0];
    return scores[topCategory] > 0 ? topCategory : 'ENTERTAINMENT'; // Default category
  }
  
  // Estimate cost level
  estimateCost(text) {
    for (const [cost, keywords] of Object.entries(this.costKeywords)) {
      for (const keyword of keywords) {
        if (text.includes(keyword)) {
          return cost;
        }
      }
    }
    return 'MEDIUM'; // Default cost
  }
  
  // Estimate time commitment
  estimateTime(text) {
    for (const [time, keywords] of Object.entries(this.timeKeywords)) {
      for (const keyword of keywords) {
        if (text.includes(keyword)) {
          return time;
        }
      }
    }
    return 'SHORT'; // Default time
  }
  
  // Extract hashtags as tags
  extractTags(post) {
    const caption = post.caption || '';
    const hashtags = caption.match(/#\w+/g) || [];
    
    // Clean and filter hashtags
    return hashtags
      .map(tag => tag.substring(1).toLowerCase())
      .filter(tag => tag.length > 2 && tag.length < 20)
      .slice(0, 10); // Limit to 10 tags
  }
  
  // Check seasonality based on date and content
  checkSeasonality(text, timestamp) {
    const date = new Date(timestamp);
    const month = date.getMonth();
    
    // Determine season from month
    let season;
    if (month >= 2 && month <= 4) season = 'SPRING';
    else if (month >= 5 && month <= 7) season = 'SUMMER';
    else if (month >= 8 && month <= 10) season = 'FALL';
    else season = 'WINTER';
    
    // Check for explicit seasonal keywords
    const seasonalKeywords = {
      'SPRING': ['spring', 'blossom', 'tulip', 'easter'],
      'SUMMER': ['summer', 'beach', 'bbq', 'outdoor', 'festival'],
      'FALL': ['fall', 'autumn', 'pumpkin', 'halloween', 'harvest'],
      'WINTER': ['winter', 'christmas', 'holiday', 'snow', 'cozy']
    };
    
    const seasons = [season]; // Start with current season
    
    for (const [seasonName, keywords] of Object.entries(seasonalKeywords)) {
      for (const keyword of keywords) {
        if (text.includes(keyword) && !seasons.includes(seasonName)) {
          seasons.push(seasonName);
        }
      }
    }
    
    return seasons;
  }
  
  // Calculate confidence score for extracted data
  calculateConfidence(factors) {
    let score = 0;
    const weights = {
      hasLocation: 40,
      hasCategory: 20,
      hasDescription: 20,
      hasTags: 20
    };
    
    for (const [factor, weight] of Object.entries(weights)) {
      if (factors[factor]) {
        score += weight;
      }
    }
    
    return score;
  }
  
  // Calculate distance between two points (simplified)
  calculateDistance(point1, point2) {
    // This is a simplified calculation
    // In a real app, you'd use a proper geolocation service
    if (!point1 || !point2) return Infinity;
    
    const lat1 = point1.latitude;
    const lon1 = point1.longitude;
    const lat2 = point2.latitude;
    const lon2 = point2.longitude;
    
    const R = 3959; // Earth's radius in miles
    const dLat = (lat2 - lat1) * Math.PI / 180;
    const dLon = (lon2 - lon1) * Math.PI / 180;
    
    const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
              Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
              Math.sin(dLon / 2) * Math.sin(dLon / 2);
    
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    return R * c;
  }
  
  // Process manual approval queue
  approveActivity(activityId) {
    const item = this.processingQueue.find(q => q.activity.id === activityId);
    if (item) {
      this.activityWheel.addActivity(item.activity);
      this.processingQueue = this.processingQueue.filter(q => q.activity.id !== activityId);
      return true;
    }
    return false;
  }
  
  // Reject activity from queue
  rejectActivity(activityId) {
    this.processingQueue = this.processingQueue.filter(q => q.activity.id !== activityId);
    return true;
  }
  
  // Get pending activities for review
  getPendingActivities() {
    return this.processingQueue.sort((a, b) => b.confidence - a.confidence);
  }
  
  // Batch process saved posts
  async processSavedPosts(instagramAPI, options = {}) {
    try {
      // Fetch saved posts from Instagram
      const savedPosts = await instagramAPI.getSavedPosts();
      
      // Filter for location-based posts
      const locationPosts = savedPosts.filter(post => 
        post.location || 
        (post.caption && this.extractLocation(post, post.caption.toLowerCase()).name)
      );
      
      // Process posts
      const results = await this.scrapeInstagramPosts(locationPosts, options);
      
      return {
        processed: results.length,
        queue: results
      };
    } catch (error) {
      console.error('Error processing saved posts:', error);
      throw error;
    }
  }
  
  // Monitor influencer posts for new activities
  async monitorInfluencers(influencerUsernames, instagramAPI, options = {}) {
    const monitoringResults = [];
    
    for (const username of influencerUsernames) {
      try {
        // Get recent posts from influencer
        const recentPosts = await instagramAPI.getUserPosts(username, { limit: 10 });
        
        // Process posts
        const results = await this.scrapeInstagramPosts(recentPosts, options);
        
        monitoringResults.push({
          influencer: username,
          newActivities: results.length,
          activities: results
        });
      } catch (error) {
        console.error(`Error monitoring ${username}:`, error);
      }
    }
    
    return monitoringResults;
  }
}

// Usage example
async function setupInstagramScraper(activityWheel, instagramAccessToken) {
  // Initialize scraper
  const scraper = new InstagramActivityScraper(activityWheel);
  
  // Initialize Instagram API wrapper
  const instagramAPI = new InstagramAPIWrapper(instagramAccessToken);
  
  // Set up influencer monitoring
  const influencers = [
    'your_favorite_food_blogger',
    'local_events_account',
    'adventure_guide_houston'
  ];
  
  // Process saved posts
  const savedResults = await scraper.processSavedPosts(instagramAPI, {
    autoAddToWheel: false,
    requireManualApproval: true,
    filterByLocation: true,
    locationRadius: 30,
    userLocation: { latitude: 29.7604, longitude: -95.3698 } // Houston coordinates
  });
  
  // Monitor influencers for new content
  const monitorResults = await scraper.monitorInfluencers(influencers, instagramAPI, {
    autoAddToWheel: false,
    requireManualApproval: true
  });
  
  return {
    scraper,
    savedResults,
    monitorResults
  };
}
