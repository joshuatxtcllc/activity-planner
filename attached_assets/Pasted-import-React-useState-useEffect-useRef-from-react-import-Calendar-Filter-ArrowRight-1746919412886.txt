import React, { useState, useEffect, useRef } from 'react';
import { 
  Calendar, Filter, ArrowRight, Check, X, Clock, 
  DollarSign, MapPin, Rotate3D, Calendar as CalendarIcon,
  Heart, Star, List, Plus, Save, Trash
} from 'lucide-react';

// Import our activity system
// In a real app you would import from your actual file path
import { 
  EnhancedActivityWheel, 
  ActivityScheduler,
  curatedActivities,
  activityCategories,
  costLevels,
  timeCommitments
} from './activity-wheel-enhancements';

const ActivityWheelUI = () => {
  // Initialize our system with curated activities
  const [activityWheel, setActivityWheel] = useState(null);
  const [scheduler, setScheduler] = useState(null);
  const [selectedActivity, setSelectedActivity] = useState(null);
  const [upcomingActivities, setUpcomingActivities] = useState([]);
  const [isSpinning, setIsSpinning] = useState(false);
  const [showFilters, setShowFilters] = useState(false);
  const [scheduleDate, setScheduleDate] = useState(new Date());
  const [notes, setNotes] = useState('');
  const [activeTab, setActiveTab] = useState('wheel');
  const [showAddActivity, setShowAddActivity] = useState(false);
  
  // New activity form state
  const [newActivity, setNewActivity] = useState({
    name: '',
    description: '',
    category: 'ENTERTAINMENT',
    costLevel: 'MEDIUM',
    timeCommitment: 'MEDIUM',
    location: '',
    seasonality: [],
    tags: []
  });
  
  // Filter state
  const [filters, setFilters] = useState({
    categories: [],
    costLevels: [],
    timeCommitments: [],
    seasons: []
  });
  
  // Tag input state
  const [tagInput, setTagInput] = useState('');
  
  // Init on mount
  useEffect(() => {
    const wheel = new EnhancedActivityWheel(curatedActivities);
    const sched = new ActivityScheduler();
    setActivityWheel(wheel);
    setScheduler(sched);
    refreshUpcomingActivities(sched);
  }, []);
  
  // Refresh upcoming activities list
  const refreshUpcomingActivities = (sched) => {
    if (!sched) return;
    const upcoming = sched.getUpcomingActivities(14); // Get next 2 weeks
    setUpcomingActivities(upcoming);
  };
  
  // Handle filter changes
  const handleFilterChange = (filterType, value) => {
    const updatedFilters = { ...filters };
    
    if (updatedFilters[filterType].includes(value)) {
      // Remove the filter if already selected
      updatedFilters[filterType] = updatedFilters[filterType].filter(item => item !== value);
    } else {
      // Add the filter
      updatedFilters[filterType] = [...updatedFilters[filterType], value];
    }
    
    setFilters(updatedFilters);
    
    // Apply filters to wheel
    if (activityWheel) {
      Object.keys(updatedFilters).forEach(key => {
        activityWheel.setFilter(key, updatedFilters[key]);
      });
    }
  };
  
  // Clear all filters
  const clearFilters = () => {
    setFilters({
      categories: [],
      costLevels: [],
      timeCommitments: [],
      seasons: []
    });
    
    if (activityWheel) {
      activityWheel.clearFilters();
    }
  };
  
  // Spin the wheel
  const spinWheel = () => {
    if (!activityWheel) return;
    
    setIsSpinning(true);
    setSelectedActivity(null);
    
    // Simulate spinning animation
    setTimeout(() => {
      const activity = activityWheel.spinWheel();
      setSelectedActivity(activity);
      setIsSpinning(false);
    }, 2000);
  };
  
  // Schedule the selected activity
  const scheduleSelectedActivity = () => {
    if (!selectedActivity || !scheduler) return;
    
    scheduler.scheduleActivity(selectedActivity, scheduleDate, notes);
    refreshUpcomingActivities(scheduler);
    
    // Reset form
    setNotes('');
    setScheduleDate(new Date());
    // Don't reset selected activity so user can see what they scheduled
  };
  
  // Handle form input for new activity
  const handleNewActivityChange = (field, value) => {
    setNewActivity(prev => ({
      ...prev,
      [field]: value
    }));
  };
  
  // Add new tag to new activity
  const handleAddTag = (e) => {
    if (e.key === 'Enter' || e.type === 'click') {
      if (!tagInput.trim() || newActivity.tags.includes(tagInput.trim())) return;
      
      setNewActivity(prev => ({
        ...prev,
        tags: [...prev.tags, tagInput.trim()]
      }));
      setTagInput('');
    }
  };
  
  // Remove tag from new activity
  const handleRemoveTag = (tagToRemove) => {
    setNewActivity(prev => ({
      ...prev,
      tags: prev.tags.filter(tag => tag !== tagToRemove)
    }));
  };
  
  // Toggle seasonality for new activity
  const handleToggleSeasonality = (season) => {
    setNewActivity(prev => {
      const seasons = [...prev.seasonality];
      if (seasons.includes(season)) {
        return {
          ...prev,
          seasonality: seasons.filter(s => s !== season)
        };
      } else {
        return {
          ...prev,
          seasonality: [...seasons, season]
        };
      }
    });
  };
  
  // Save new activity
  const saveNewActivity = () => {
    if (!newActivity.name || !activityWheel) return;
    
    // Create new activity instance and add to wheel
    const activity = {
      ...newActivity,
      isUserAdded: true
    };
    
    activityWheel.addActivity(activity);
    
    // Reset form and close modal
    setNewActivity({
      name: '',
      description: '',
      category: 'ENTERTAINMENT',
      costLevel: 'MEDIUM',
      timeCommitment: 'MEDIUM',
      location: '',
      seasonality: [],
      tags: []
    });
    
    setShowAddActivity(false);
  };
  
  // Mark an activity as complete
  const completeActivity = (activityId) => {
    if (!scheduler) return;
    
    scheduler.completeActivity(activityId);
    refreshUpcomingActivities(scheduler);
  };
  
  // UI Components
  
  // Filter badge component
  const FilterBadge = ({ type, id, label, isSelected, onClick }) => (
    <button
      className={`px-3 py-1 rounded-full text-sm flex items-center gap-1 transition-colors ${
        isSelected ? 'bg-blue-500 text-white' : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
      }`}
      onClick={() => onClick(type, id)}
    >
      {label}
      {isSelected && <Check size={14} />}
    </button>
  );
  
  // Filter section component
  const FilterSection = ({ title, items, type, selectedItems }) => (
    <div className="mb-4">
      <h3 className="font-medium mb-2">{title}</h3>
      <div className="flex flex-wrap gap-2">
        {Object.entries(items).map(([id, item]) => (
          <FilterBadge
            key={id}
            type={type}
            id={id}
            label={item.label}
            isSelected={selectedItems.includes(id)}
            onClick={handleFilterChange}
          />
        ))}
      </div>
    </div>
  );
  
  // Activity card component
  const ActivityCard = ({ activity, onSchedule, isSelected = false }) => (
    <div className={`border rounded-lg p-4 ${isSelected ? 'border-blue-500 bg-blue-50' : 'border-gray-200'}`}>
      <div className="flex justify-between items-start mb-2">
        <h3 className="font-semibold text-lg">{activity.name}</h3>
        <button 
          onClick={() => onSchedule(activity)}
          className="text-blue-500 hover:text-blue-700"
        >
          <CalendarIcon size={20} />
        </button>
      </div>
      
      <p className="text-gray-600 text-sm mb-3">{activity.description}</p>
      
      <div className="flex flex-wrap gap-2 text-sm text-gray-500">
        <span className="flex items-center gap-1">
          <MapPin size={14} /> {activity.location}
        </span>
        <span className="flex items-center gap-1">
          <Clock size={14} /> {timeCommitments[activity.timeCommitment]?.label}
        </span>
        <span className="flex items-center gap-1">
          <DollarSign size={14} /> {costLevels[activity.costLevel]?.label}
        </span>
      </div>
      
      {activity.tags.length > 0 && (
        <div className="mt-3 flex flex-wrap gap-1">
          {activity.tags.map(tag => (
            <span key={tag} className="px-2 py-0.5 bg-gray-100 rounded text-xs">
              {tag}
            </span>
          ))}
        </div>
      )}
    </div>
  );
  
  // Navigation tabs
  const tabs = [
    { id: 'wheel', label: 'Spin Wheel', icon: Rotate3D },
    { id: 'schedule', label: 'Schedule', icon: Calendar },
    { id: 'activities', label: 'Activities', icon: List }
  ];
  
  // Get date string for input
  const getDateString = (date) => {
    return date.toISOString().split('T')[0];
  };
  
  return (
    <div className="max-w-6xl mx-auto p-4">
      {/* Navigation */}
      <div className="flex justify-between items-center mb-6">
        <div className="flex gap-1 bg-gray-100 rounded-lg p-1">
          {tabs.map(tab => (
            <button
              key={tab.id}
              onClick={() => setActiveTab(tab.id)}
              className={`flex items-center gap-2 px-4 py-2 rounded transition-colors ${
                activeTab === tab.id 
                  ? 'bg-white text-blue-600 shadow-sm' 
                  : 'text-gray-600 hover:text-gray-900'
              }`}
            >
              <tab.icon size={16} />
              {tab.label}
            </button>
          ))}
        </div>
        
        <button
          onClick={() => setShowAddActivity(true)}
          className="flex items-center gap-2 px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
        >
          <Plus size={16} />
          Add Activity
        </button>
      </div>
      
      {/* Content */}
      {activeTab === 'wheel' && (
        <div className="space-y-6">
          {/* Filters */}
          <div className="bg-white rounded-lg border p-4">
            <button
              onClick={() => setShowFilters(!showFilters)}
              className="flex items-center gap-2 mb-4 text-gray-700 hover:text-gray-900"
            >
              <Filter size={16} />
              Filters
              {showFilters ? <X size={14} /> : <ArrowRight size={14} />}
            </button>
            
            {showFilters && (
              <div>
                <FilterSection 
                  title="Categories" 
                  items={activityCategories} 
                  type="categories" 
                  selectedItems={filters.categories} 
                />
                <FilterSection 
                  title="Cost" 
                  items={costLevels} 
                  type="costLevels" 
                  selectedItems={filters.costLevels} 
                />
                <FilterSection 
                  title="Time" 
                  items={timeCommitments} 
                  type="timeCommitments" 
                  selectedItems={filters.timeCommitments} 
                />
                
                <button
                  onClick={clearFilters}
                  className="mt-4 text-sm text-gray-500 hover:text-gray-700"
                >
                  Clear all filters
                </button>
              </div>
            )}
          </div>
          
          {/* Wheel */}
          <div className="bg-white rounded-lg border p-8 text-center">
            <button
              onClick={spinWheel}
              disabled={isSpinning}
              className={`inline-flex items-center gap-2 px-8 py-4 rounded-lg text-lg font-medium transition-all ${
                isSpinning 
                  ? 'bg-gray-200 text-gray-500 cursor-not-allowed' 
                  : 'bg-blue-500 text-white hover:bg-blue-600 active:scale-95'
              }`}
            >
              <Rotate3D size={24} className={isSpinning ? 'animate-spin' : ''} />
              {isSpinning ? 'Spinning...' : 'Spin the Wheel!'}
            </button>
            
            {activityWheel && (
              <p className="mt-4 text-sm text-gray-500">
                {activityWheel.getActivityCount()} activities available
              </p>
            )}
          </div>
          
          {/* Selected Activity */}
          {selectedActivity && (
            <div className="bg-white rounded-lg border p-6">
              <h2 className="text-xl font-semibold mb-4">Your Next Activity:</h2>
              <ActivityCard 
                activity={selectedActivity} 
                onSchedule={() => {}} 
                isSelected={true} 
              />
              
              {/* Schedule form */}
              <div className="mt-6 flex gap-4">
                <input
                  type="date"
                  value={getDateString(scheduleDate)}
                  onChange={(e) => setScheduleDate(new Date(e.target.value))}
                  className="border rounded px-3 py-2"
                />
                <input
                  type="text"
                  placeholder="Add notes..."
                  value={notes}
                  onChange={(e) => setNotes(e.target.value)}
                  className="flex-1 border rounded px-3 py-2"
                />
                <button
                  onClick={scheduleSelectedActivity}
                  className="px-4 py-2 bg-green-500 text-white rounded hover:bg-green-600"
                >
                  Schedule
                </button>
              </div>
            </div>
          )}
        </div>
      )}
      
      {activeTab === 'schedule' && (
        <div className="bg-white rounded-lg border overflow-hidden">
          <div className="border-b p-4 bg-gray-50">
            <h2 className="font-semibold">Upcoming Activities</h2>
          </div>
          
          <div className="divide-y">
            {upcomingActivities.length === 0 ? (
              <div className="p-8 text-center text-gray-500">
                No activities scheduled yet. Click "Spin the Wheel" to get started!
              </div>
            ) : (
              upcomingActivities.map(activity => (
                <div key={activity.id} className="p-4 flex justify-between items-center">
                  <div>
                    <h3 className="font-medium">{activity.activityName}</h3>
                    <div className="text-sm text-gray-500">
                      {new Date(activity.scheduledDate).toLocaleDateString()} 
                      {activity.notes && ` - ${activity.notes}`}
                    </div>
                  </div>
                  
                  <button
                    onClick={() => completeActivity(activity.id)}
                    className="text-green-500 hover:text-green-700"
                  >
                    <Check size={20} />
                  </button>
                </div>
              ))
            )}
          </div>
        </div>
      )}
      
      {activeTab === 'activities' && (
        <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
          {activityWheel?.filteredActivities.map(activity => (
            <ActivityCard 
              key={activity.id} 
              activity={activity} 
              onSchedule={() => {}} 
            />
          ))}
        </div>
      )}
      
      {/* Add Activity Modal */}
      {showAddActivity && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50">
          <div className="bg-white rounded-lg max-w-md w-full p-6">
            <div className="flex justify-between items-center mb-4">
              <h2 className="text-xl font-semibold">Add New Activity</h2>
              <button
                onClick={() => setShowAddActivity(false)}
                className="text-gray-500 hover:text-gray-700"
              >
                <X size={20} />
              </button>
            </div>
            
            <form className="space-y-4">
              <div>
                <label className="block text-sm font-medium mb-1">Name</label>
                <input
                  type="text"
                  value={newActivity.name}
                  onChange={(e) => handleNewActivityChange('name', e.target.value)}
                  className="w-full border rounded px-3 py-2"
                  required
                />
              </div>
              
              <div>
                <label className="block text-sm font-medium mb-1">Description</label>
                <textarea
                  value={newActivity.description}
                  onChange={(e) => handleNewActivityChange('description', e.target.value)}
                  className="w-full border rounded px-3 py-2"
                  rows={3}
                />
              </div>
              
              <div>
                <label className="block text-sm font-medium mb-1">Location</label>
                <input
                  type="text"
                  value={newActivity.location}
                  onChange={(e) => handleNewActivityChange('location', e.target.value)}
                  className="w-full border rounded px-3 py-2"
                />
              </div>
              
              <div className="grid grid-cols-3 gap-4">
                <div>
                  <label className="block text-sm font-medium mb-1">Category</label>
                  <select
                    value={newActivity.category}
                    onChange={(e) => handleNewActivityChange('category', e.target.value)}
                    className="w-full border rounded px-3 py-2"
                  >
                    {Object.entries(activityCategories).map(([id, cat]) => (
                      <option key={id} value={id}>{cat.label}</option>
                    ))}
                  </select>
                </div>
                
                <div>
                  <label className="block text-sm font-medium mb-1">Cost</label>
                  <select
                    value={newActivity.costLevel}
                    onChange={(e) => handleNewActivityChange('costLevel', e.target.value)}
                    className="w-full border rounded px-3 py-2"
                  >
                    {Object.entries(costLevels).map(([id, cost]) => (
                      <option key={id} value={id}>{cost.label}</option>
                    ))}
                  </select>
                </div>
                
                <div>
                  <label className="block text-sm font-medium mb-1">Time</label>
                  <select
                    value={newActivity.timeCommitment}
                    onChange={(e) => handleNewActivityChange('timeCommitment', e.target.value)}
                    className="w-full border rounded px-3 py-2"
                  >
                    {Object.entries(timeCommitments).map(([id, time]) => (
                      <option key={id} value={id}>{time.label}</option>
                    ))}
                  </select>
                </div>
              </div>
              
              <div>
                <label className="block text-sm font-medium mb-1">Seasons (optional)</label>
                <div className="flex gap-2">
                  {['SPRING', 'SUMMER', 'FALL', 'WINTER'].map(season => (
                    <button
                      key={season}
                      type="button"
                      onClick={() => handleToggleSeasonality(season)}
                      className={`px-3 py-1 rounded text-sm ${
                        newActivity.seasonality.includes(season)
                          ? 'bg-blue-500 text-white'
                          : 'bg-gray-100 text-gray-700'
                      }`}
                    >
                      {season}
                    </button>
                  ))}
                </div>
              </div>
              
              <div>
                <label className="block text-sm font-medium mb-1">Tags</label>
                <div className="flex gap-2 mb-2">
                  <input
                    type="text"
                    value={tagInput}
                    onChange={(e) => setTagInput(e.target.value)}
                    onKeyDown={handleAddTag}
                    placeholder="Add tags..."
                    className="flex-1 border rounded px-3 py-2"
                  />
                  <button
                    type="button"
                    onClick={handleAddTag}
                    className="px-3 py-2 bg-gray-100 text-gray-700 rounded hover:bg-gray-200"
                  >
                    Add
                  </button>
                </div>
                
                <div className="flex flex-wrap gap-1">
                  {newActivity.tags.map(tag => (
                    <span 
                      key={tag} 
                      className="flex items-center gap-1 px-2 py-1 bg-blue-100 text-blue-800 rounded text-sm"
                    >
                      {tag}
                      <button
                        type="button"
                        onClick={() => handleRemoveTag(tag)}
                        className="hover:text-blue-900"
                      >
                        <X size={12} />
                      </button>
                    </span>
                  ))}
                </div>
              </div>
              
              <div className="flex gap-4 pt-4">
                <button
                  type="button"
                  onClick={() => setShowAddActivity(false)}
                  className="flex-1 px-4 py-2 border rounded text-gray-700 hover:bg-gray-50"
                >
                  Cancel
                </button>
                <button
                  type="button"
                  onClick={saveNewActivity}
                  className="flex-1 px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
                  disabled={!newActivity.name}
                >
                  Save Activity
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};

export default ActivityWheelUI;
