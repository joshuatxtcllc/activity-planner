import React, { useState, useEffect } from 'react';
import { 
  Instagram, CheckCircle, XCircle, MapPin, Clock, DollarSign, Tag,
  RefreshCw, Filter, Search, Calendar, AlertCircle, Download, Settings,
  Plus, Trash, Users, Bookmark, Eye
} from 'lucide-react';

// Import the scraper
import { InstagramActivityScraper, InstagramAPIWrapper } from './instagram-activity-scraper';

const InstagramActivityReviewer = ({ activityWheel, onActivitiesAdded }) => {
  const [scraper, setScraper] = useState(null);
  const [isConnected, setIsConnected] = useState(false);
  const [pendingActivities, setPendingActivities] = useState([]);
  const [savedPosts, setSavedPosts] = useState([]);
  const [influencers, setInfluencers] = useState([]);
  const [newInfluencer, setNewInfluencer] = useState('');
  const [isProcessing, setIsProcessing] = useState(false);
  const [filterText, setFilterText] = useState('');
  const [selectedCategory, setSelectedCategory] = useState('ALL');
  const [settings, setSettings] = useState({
    autoAddToWheel: false,
    requireManualApproval: true,
    filterByLocation: true,
    locationRadius: 30,
    userLocation: null
  });
  const [activeTab, setActiveTab] = useState('pending');
  const [showSettings, setShowSettings] = useState(false);
  
  // Initialize scraper
  useEffect(() => {
    if (activityWheel) {
      const newScraper = new InstagramActivityScraper(activityWheel);
      setScraper(newScraper);
    }
  }, [activityWheel]);
  
  // Connect to Instagram
  const connectInstagram = async () => {
    // This would open Instagram OAuth flow
    // For demo purposes, we'll simulate connection
    setIsConnected(true);
    
    // Load saved influencers from localStorage
    const savedInfluencers = localStorage.getItem('instagram_influencers');
    if (savedInfluencers) {
      setInfluencers(JSON.parse(savedInfluencers));
    }
  };
  
  // Process saved posts
  const processSavedPosts = async () => {
    if (!scraper || !isConnected) return;
    
    setIsProcessing(true);
    try {
      // Simulate fetching saved posts
      const mockSavedPosts = [
        {
          id: '1',
          caption: 'Just tried the amazing sushi at Kata Robata! Must order the omakase - totally worth the price! 📍 Houston #sushi #omakase',
          location: { name: 'Kata Robata', coordinates: { latitude: 29.7623, longitude: -95.3639 } },
          timestamp: new Date().toISOString(),
          imageUrl: '/api/placeholder/300/300',
          url: 'https://instagram.com/p/example1'
        },
        {
          id: '2',
          caption: 'Perfect hiking spot at Buffalo Bayou Park! Free and beautiful views of downtown. Go early to beat the heat! 🌳 #houston #hiking',
          location: { name: 'Buffalo Bayou Park', coordinates: { latitude: 29.7633, longitude: -95.3906 } },
          timestamp: new Date().toISOString(),
          imageUrl: '/api/placeholder/300/300',
          url: 'https://instagram.com/p/example2'
        }
      ];
      
      // Process posts through scraper
      const results = await scraper.scrapeInstagramPosts(mockSavedPosts, settings);
      setPendingActivities(scraper.getPendingActivities());
      
    } catch (error) {
      console.error('Error processing saved posts:', error);
      alert('Error processing saved posts. Please try again.');
    } finally {
      setIsProcessing(false);
    }
  };
  
  // Add new influencer
  const addInfluencer = () => {
    if (!newInfluencer.trim() || influencers.includes(newInfluencer)) return;
    
    const updatedInfluencers = [...influencers, newInfluencer.trim()];
    setInfluencers(updatedInfluencers);
    localStorage.setItem('instagram_influencers', JSON.stringify(updatedInfluencers));
    setNewInfluencer('');
  };
  
  // Remove influencer
  const removeInfluencer = (username) => {
    const updatedInfluencers = influencers.filter(inf => inf !== username);
    setInfluencers(updatedInfluencers);
    localStorage.setItem('instagram_influencers', JSON.stringify(updatedInfluencers));
  };
  
  // Monitor influencers
  const monitorInfluencers = async () => {
    if (!scraper || !isConnected || influencers.length === 0) return;
    
    setIsProcessing(true);
    try {
      // Simulate monitoring influencers
      const mockInfluencerPosts = influencers.map(username => ({
        influencer: username,
        posts: [
          {
            id: `${username}_1`,
            caption: `New spot alert! Check out ${username}'s latest recommendation...`,
            location: { name: 'Trendy New Spot', coordinates: { latitude: 29.7604, longitude: -95.3698 } },
            timestamp: new Date().toISOString(),
            imageUrl: '/api/placeholder/300/300',
            url: `https://instagram.com/p/${username}_example`
          }
        ]
      }));
      
      // Process each influencer's posts
      for (const { posts } of mockInfluencerPosts) {
        await scraper.scrapeInstagramPosts(posts, settings);
      }
      
      setPendingActivities(scraper.getPendingActivities());
      
    } catch (error) {
      console.error('Error monitoring influencers:', error);
      alert('Error monitoring influencers. Please try again.');
    } finally {
      setIsProcessing(false);
    }
  };
  
  // Approve an activity
  const approveActivity = (activityId) => {
    if (!scraper) return;
    
    scraper.approveActivity(activityId);
    setPendingActivities(scraper.getPendingActivities());
    
    if (onActivitiesAdded) {
      onActivitiesAdded(1);
    }
  };
  
  // Reject an activity
  const rejectActivity = (activityId) => {
    if (!scraper) return;
    
    scraper.rejectActivity(activityId);
    setPendingActivities(scraper.getPendingActivities());
  };
  
  // Approve all activities
  const approveAll = () => {
    pendingActivities.forEach(item => {
      scraper.approveActivity(item.activity.id);
    });
    setPendingActivities([]);
    
    if (onActivitiesAdded) {
      onActivitiesAdded(pendingActivities.length);
    }
  };
  
  // Filter pending activities
  const filteredActivities = pendingActivities.filter(item => {
    const matchesText = item.activity.name.toLowerCase().includes(filterText.toLowerCase()) ||
                       item.activity.location.toLowerCase().includes(filterText.toLowerCase());
    const matchesCategory = selectedCategory === 'ALL' || item.activity.category === selectedCategory;
    
    return matchesText && matchesCategory;
  });
  
  // Activity card component
  const ActivityCard = ({ item }) => {
    const { activity, confidence, post } = item;
    
    const getConfidenceColor = (conf) => {
      if (conf >= 80) return 'text-green-600 bg-green-100';
      if (conf >= 60) return 'text-yellow-600 bg-yellow-100';
      return 'text-red-600 bg-red-100';
    };
    
    return (
      <div className="border rounded-lg p-4 space-y-3">
        <div className="flex justify-between items-start">
          <div className="flex-1">
            <div className="flex items-center gap-2 mb-1">
              <h3 className="font-semibold">{activity.name}</h3>
              <span className={`px-2 py-0.5 rounded text-xs font-medium ${getConfidenceColor(confidence)}`}>
                {confidence}% confident
              </span>
            </div>
            <p className="text-sm text-gray-600">{activity.description}</p>
          </div>
          
          <div className="flex gap-2">
            <button
              onClick={() => approveActivity(activity.id)}
              className="p-2 text-green-600 hover:bg-green-50 rounded"
              title="Approve"
            >
              <CheckCircle size={20} />
            </button>
            <button
              onClick={() => rejectActivity(activity.id)}
              className="p-2 text-red-600 hover:bg-red-50 rounded"
              title="Reject"
            >
              <XCircle size={20} />
            </button>
          </div>
        </div>
        
        <div className="flex flex-wrap gap-2 text-sm text-gray-500">
          <span className="flex items-center gap-1">
            <MapPin size={14} /> {activity.location}
          </span>
          <span className="flex items-center gap-1">
            <Clock size={14} /> {activity.timeCommitment}
          </span>
          <span className="flex items-center gap-1">
            <DollarSign size={14} /> {activity.costLevel}
          </span>
          <span className="flex items-center gap-1">
            <Tag size={14} /> {activity.category}
          </span>
        </div>
        
        {activity.tags.length > 0 && (
          <div className="flex flex-wrap gap-1">
            {activity.tags.map(tag => (
              <span key={tag} className="px-2 py-0.5 bg-gray-100 rounded text-xs">
                #{tag}
              </span>
            ))}
          </div>
        )}
        
        <div className="flex justify-between items-center pt-2 border-t text-xs text-gray-400">
          <span>From @{post.poster || 'unknown'}</span>
          <a 
            href={post.url} 
            target="_blank" 
            rel="noopener noreferrer"
            className="text-blue-500 hover:text-blue-700"
          >
            View original post
          </a>
        </div>
      </div>
    );
  };
  
  if (!isConnected) {
    return (
      <div className="max-w-2xl mx-auto p-6">
        <div className="bg-white rounded-lg border p-8 text-center">
          <Instagram size={48} className="mx-auto mb-4 text-gray-400" />
          <h2 className="text-xl font-semibold mb-2">Connect Instagram</h2>
          <p className="text-gray-600 mb-6">
            Connect your Instagram account to automatically discover and add activities from your saved posts and favorite influencers.
          </p>
          <button
            onClick={connectInstagram}
            className="bg-gradient-to-r from-purple-500 to-pink-500 text-white px-6 py-3 rounded-lg hover:opacity-90 transition-opacity"
          >
            Connect with Instagram
          </button>
        </div>
      </div>
    );
  }
  
  // Tabs
  const tabs = [
    { id: 'pending', label: 'Pending Activities', icon: AlertCircle, count: pendingActivities.length },
    { id: 'influencers', label: 'Influencers', icon: Users, count: influencers.length },
    { id: 'saved', label: 'Saved Posts', icon: Bookmark, count: savedPosts.length }
  ];
  
  return (
    <div className="max-w-6xl mx-auto p-4">
      {/* Header */}
      <div className="flex justify-between items-center mb-6">
        <div className="flex items-center gap-3">
          <Instagram size={24} className="text-pink-500" />
          <h1 className="text-xl font-semibold">Instagram Activities</h1>
        </div>
        <button
          onClick={() => setShowSettings(!showSettings)}
          className="p-2 text-gray-600 hover:bg-gray-100 rounded"
        >
          <Settings size={20} />
        </button>
      </div>
      
      {/* Settings Panel */}
      {showSettings && (
        <div className="mb-6 p-4 bg-gray-50 rounded-lg">
          <h3 className="font-medium mb-3">Import Settings</h3>
          <div className="space-y-3">
            <label className="flex items-center gap-2">
              <input
                type="checkbox"
                checked={settings.requireManualApproval}
                onChange={(e) => setSettings({...settings, requireManualApproval: e.target.checked})}
                className="rounded"
              />
              Require manual approval for activities
            </label>
            <label className="flex items-center gap-2">
              <input
                type="checkbox"
                checked={settings.filterByLocation}
                onChange={(e) => setSettings({...settings, filterByLocation: e.target.checked})}
                className="rounded"
              />
              Filter activities by location
            </label>
            {settings.filterByLocation && (
              <div className="ml-6">
                <label className="block text-sm text-gray-600 mb-1">
                  Distance radius (miles)
                </label>
                <input
                  type="number"
                  value={settings.locationRadius}
                  onChange={(e) => setSettings({...settings, locationRadius: parseInt(e.target.value)})}
                  className="w-24 px-3 py-1 border rounded"
                  min="1"
                  max="100"
                />
              </div>
            )}
          </div>
        </div>
      )}
      
      {/* Tabs */}
      <div className="flex gap-1 border-b mb-6">
        {tabs.map(tab => (
          <button
            key={tab.id}
            onClick={() => setActiveTab(tab.id)}
            className={`flex items-center gap-2 px-4 py-2 -mb-px ${
              activeTab === tab.id
                ? 'border-b-2 border-blue-500 text-blue-600'
                : 'text-gray-600 hover:text-gray-900'
            }`}
          >
            <tab.icon size={16} />
            {tab.label}
            {tab.count > 0 && (
              <span className="bg-blue-100 text-blue-600 px-2 py-0.5 rounded-full text-xs">
                {tab.count}
              </span>
            )}
          </button>
        ))}
      </div>
      
      {/* Content */}
      {activeTab === 'pending' && (
        <div className="space-y-4">
          {/* Actions Bar */}
          <div className="flex justify-between items-center">
            <div className="flex gap-2">
              <div className="relative flex-1 max-w-md">
                <Search size={16} className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400" />
                <input
                  type="text"
                  placeholder="Search activities..."
                  value={filterText}
                  onChange={(e) => setFilterText(e.target.value)}
                  className="pl-10 pr-4 py-2 border rounded-lg w-full"
                />
              </div>
              <select
                value={selectedCategory}
                onChange={(e) => setSelectedCategory(e.target.value)}
                className="px-4 py-2 border rounded-lg"
              >
                <option value="ALL">All Categories</option>
                <option value="FOOD_DRINK">Food & Drink</option>
                <option value="OUTDOOR">Outdoor</option>
                <option value="ENTERTAINMENT">Entertainment</option>
                <option value="CULTURE">Culture</option>
                <option value="ACTIVE">Active</option>
              </select>
            </div>
            
            <div className="flex gap-2">
              <button
                onClick={processSavedPosts}
                disabled={isProcessing}
                className="flex items-center gap-2 px-4 py-2 border rounded-lg hover:bg-gray-50 disabled:opacity-50"
              >
                <Bookmark size={16} />
                {isProcessing ? <RefreshCw size={16} className="animate-spin" /> : 'Process Saved'}
              </button>
              <button
                onClick={monitorInfluencers}
                disabled={isProcessing || influencers.length === 0}
                className="flex items-center gap-2 px-4 py-2 border rounded-lg hover:bg-gray-50 disabled:opacity-50"
              >
                <Eye size={16} />
                {isProcessing ? <RefreshCw size={16} className="animate-spin" /> : 'Check Influencers'}
              </button>
              {pendingActivities.length > 0 && (
                <button
                  onClick={approveAll}
                  className="flex items-center gap-2 px-4 py-2 bg-green-500 text-white rounded-lg hover:bg-green-600"
                >
                  <CheckCircle size={16} />
                  Approve All
                </button>
              )}
            </div>
          </div>
          
          {/* Activities List */}
          {isProcessing ? (
            <div className="text-center py-12">
              <RefreshCw size={32} className="mx-auto mb-3 animate-spin text-gray-400" />
              <p className="text-gray-600">Processing Instagram posts...</p>
            </div>
          ) : filteredActivities.length === 0 ? (
            <div className="text-center py-12 text-gray-500">
              {pendingActivities.length === 0 ? 
                'No pending activities. Process saved posts or check influencers to find new activities!' :
                'No activities match your search criteria.'
              }
            </div>
          ) : (
            <div className="grid gap-4">
              {filteredActivities.map(item => (
                <ActivityCard key={item.activity.id} item={item} />
              ))}
            </div>
          )}
        </div>
      )}
      
      {activeTab === 'influencers' && (
        <div className="space-y-4">
          <div className="flex gap-2">
            <input
              type="text"
              placeholder="Add influencer username..."
              value={newInfluencer}
              onChange={(e) => setNewInfluencer(e.target.value)}
              onKeyDown={(e) => e.key === 'Enter' && addInfluencer()}
              className="flex-1 px-4 py-2 border rounded-lg"
            />
            <button
              onClick={addInfluencer}
              className="flex items-center gap-2 px-4 py-2 bg-blue-500 text-white rounded-lg hover:bg-blue-600"
            >
              <Plus size={16} />
              Add
            </button>
          </div>
          
          <div className="grid gap-2">
            {influencers.length === 0 ? (
              <div className="text-center py-12 text-gray-500">
                No influencers added yet. Add usernames above to monitor their posts.
              </div>
            ) : (
              influencers.map(username => (
                <div key={username} className="flex items-center justify-between p-4 border rounded-lg">
                  <div className="flex items-center gap-3">
                    <Instagram size={20} className="text-pink-500" />
                    <span className="font-medium">@{username}</span>
                  </div>
                  <button
                    onClick={() => removeInfluencer(username)}
                    className="p-2 text-red-600 hover:bg-red-50 rounded"
                  >
                    <Trash size={16} />
                  </button>
                </div>
              ))
            )}
          </div>
        </div>
      )}
      
      {activeTab === 'saved' && (
        <div className="space-y-4">
          <div className="text-center py-12 text-gray-500">
            Saved posts will appear here when processed. Click "Process Saved" above to scan your saved Instagram posts.
          </div>
        </div>
      )}
    </div>
  );
};

export default InstagramActivityReviewer;
