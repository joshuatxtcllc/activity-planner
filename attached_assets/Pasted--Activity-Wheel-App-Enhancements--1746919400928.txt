// 2. ENHANCED ACTIVITY MODEL
// ============================================================

class EnhancedActivity {
  constructor({
    id,
    name,
    description,
    category,
    costLevel,
    timeCommitment,
    location,
    isUserAdded = true,
    eventbriteId = null,
    ticketmasterId = null,
    seasonality = [], // e.g., ['SPRING', 'SUMMER']
    imageUrl = '',
    url = '',
    contactInfo = '',
    rating = 0,
    tags = []
  }) {
    this.id = id || generateUniqueId();
    this.name = name;
    this.description = description;
    this.category = category;
    this.costLevel = costLevel;
    this.timeCommitment = timeCommitment;
    this.location = location;
    this.isUserAdded = isUserAdded;
    this.eventbriteId = eventbriteId;
    this.ticketmasterId = ticketmasterId;
    this.seasonality = seasonality;
    this.imageUrl = imageUrl;
    this.url = url;
    this.contactInfo = contactInfo;
    this.rating = rating;
    this.tags = tags;
    this.dateAdded = new Date();
    this.lastSelected = null;
    this.timesSelected = 0;
  }

  markAsSelected() {
    this.lastSelected = new Date();
    this.timesSelected += 1;
  }

  // Check if activity is appropriate for current season
  isInSeason() {
    if (this.seasonality.length === 0) return true; // No seasonality restrictions
    
    const currentSeason = getCurrentSeason();
    return this.seasonality.includes(currentSeason);
  }
}

// Helper function to determine current season
function getCurrentSeason() {
  const month = new Date().getMonth();
  if (month >= 2 && month <= 4) return 'SPRING';
  if (month >= 5 && month <= 7) return 'SUMMER';
  if (month >= 8 && month <= 10) return 'FALL';
  return 'WINTER';
}

// Helper function to generate unique ID
function generateUniqueId() {
  return Date.now().toString(36) + Math.random().toString(36).substr(2, 5);
}

// 3. WEEKLY SCHEDULER IMPLEMENTATION
// ============================================================

class ActivityScheduler {
  constructor() {
    this.scheduledActivities = [];
    this.reminderSettings = {
      enabled: true,
      daysBefore: 1,
      notificationType: 'push' // or 'email'
    };
  }

  // Schedule an activity
  scheduleActivity(activity, date, notes = '') {
    const scheduledActivity = {
      id: generateUniqueId(),
      activityId: activity.id,
      activityName: activity.name,
      scheduledDate: date,
      notes: notes,
      isCompleted: false,
      reminderSent: false
    };
    
    this.scheduledActivities.push(scheduledActivity);
    this.sortSchedule();
    
    if (this.reminderSettings.enabled) {
      this.scheduleReminder(scheduledActivity);
    }
    
    return scheduledActivity;
  }
  
  // Get upcoming activities
  getUpcomingActivities(days = 7) {
    const now = new Date();
    const futureDate = new Date();
    futureDate.setDate(now.getDate() + days);
    
    return this.scheduledActivities.filter(item => {
      const itemDate = new Date(item.scheduledDate);
      return itemDate >= now && itemDate <= futureDate && !item.isCompleted;
    });
  }
  
  // Mark activity as completed
  completeActivity(scheduledActivityId, rating = 0, review = '') {
    const activity = this.scheduledActivities.find(a => a.id === scheduledActivityId);
    if (activity) {
      activity.isCompleted = true;
      activity.completedDate = new Date();
      activity.rating = rating;
      activity.review = review;
      return true;
    }
    return false;
  }
  
  // Helper to sort schedule by date
  sortSchedule() {
    this.scheduledActivities.sort((a, b) => {
      return new Date(a.scheduledDate) - new Date(b.scheduledDate);
    });
  }
  
  // Schedule a reminder for an activity
  scheduleReminder(scheduledActivity) {
    // Implementation depends on your notification system
    // This is a placeholder for the concept
    const reminderDate = new Date(scheduledActivity.scheduledDate);
    reminderDate.setDate(reminderDate.getDate() - this.reminderSettings.daysBefore);
    
    // Here you would integrate with a notification service
    console.log(`Reminder set for ${scheduledActivity.activityName} on ${reminderDate.toLocaleDateString()}`);
  }
  
  // Get statistics about completed activities
  getActivityStats() {
    const completed = this.scheduledActivities.filter(a => a.isCompleted);
    
    if (completed.length === 0) {
      return {
        totalCompleted: 0,
        averageRating: 0,
        categoryBreakdown: {},
        costBreakdown: {}
      };
    }
    
    // This would need to be integrated with your activity database
    // to pull full activity details for analysis
    return {
      totalCompleted: completed.length,
      averageRating: completed.reduce((sum, a) => sum + (a.rating || 0), 0) / completed.length,
      // Additional statistics would be calculated here
    };
  }
}

// 5. WHEEL ENHANCEMENT - FILTERING IMPLEMENTATION
// ============================================================

class EnhancedActivityWheel {
  constructor(activities = []) {
    this.allActivities = activities;
    this.filteredActivities = [...activities];
    this.filters = {
      categories: [],
      costLevels: [],
      timeCommitments: [],
      seasons: [],
      tags: []
    };
  }
  
  // Add activity
  addActivity(activity) {
    this.allActivities.push(activity);
    this.applyFilters(); // Re-apply filters to update filtered list
  }
  
  // Add multiple activities
  addActivities(activities) {
    this.allActivities = [...this.allActivities, ...activities];
    this.applyFilters();
  }
  
  // Set filter
  setFilter(filterType, values) {
    if (this.filters.hasOwnProperty(filterType)) {
      this.filters[filterType] = values;
      this.applyFilters();
    }
  }
  
  // Clear all filters
  clearFilters() {
    for (const key in this.filters) {
      this.filters[key] = [];
    }
    this.filteredActivities = [...this.allActivities];
  }
  
  // Apply all current filters
  applyFilters() {
    this.filteredActivities = this.allActivities.filter(activity => {
      // Category filter
      if (this.filters.categories.length > 0 && !this.filters.categories.includes(activity.category)) {
        return false;
      }
      
      // Cost level filter
      if (this.filters.costLevels.length > 0 && !this.filters.costLevels.includes(activity.costLevel)) {
        return false;
      }
      
      // Time commitment filter
      if (this.filters.timeCommitments.length > 0 && !this.filters.timeCommitments.includes(activity.timeCommitment)) {
        return false;
      }
      
      // Season filter
      if (this.filters.seasons.length > 0) {
        if (activity.seasonality.length === 0) {
          // Activity has no seasonal restrictions
          if (this.filters.seasons.includes('ANY')) {
            return true;
          }
        } else if (!activity.seasonality.some(s => this.filters.seasons.includes(s))) {
          return false;
        }
      }
      
      // Tags filter (match any tag)
      if (this.filters.tags.length > 0 && !activity.tags.some(tag => this.filters.tags.includes(tag))) {
        return false;
      }
      
      return true;
    });
  }
  
  // Get activity count
  getActivityCount() {
    return this.filteredActivities.length;
  }
  
  // Spin the wheel and get random activity
  spinWheel() {
    if (this.filteredActivities.length === 0) {
      return null;
    }
    
    const randomIndex = Math.floor(Math.random() * this.filteredActivities.length);
    const selectedActivity = this.filteredActivities[randomIndex];
    
    // Update activity statistics
    const actualActivity = this.allActivities.find(a => a.id === selectedActivity.id);
    if (actualActivity) {
      actualActivity.markAsSelected();
    }
    
    return selectedActivity;
  }
  
  // Get activities that haven't been selected recently
  getFreshActivities(days = 30) {
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - days);
    
    return this.filteredActivities.filter(activity => 
      !activity.lastSelected || new Date(activity.lastSelected) < cutoffDate
    );
  }
  
  // Get most selected activities
  getMostSelectedActivities(limit = 10) {
    return [...this.allActivities]
      .sort((a, b) => b.timesSelected - a.timesSelected)
      .slice(0, limit);
  }
}

// 6. USAGE EXAMPLE
// ============================================================

// Initialize the system
function initializeActivitySystem() {
  // Create the wheel with curated activities
  const activityWheel = new EnhancedActivityWheel(curatedActivities);
  
  // Create the scheduler
  const scheduler = new ActivityScheduler();
  
  // Example usage
  console.log(`Loaded ${activityWheel.getActivityCount()} curated activities`);
  
  // Spin for weekend activity with outdoor filter
  activityWheel.setFilter('categories', ['OUTDOOR']);
  const weekendActivity = activityWheel.spinWheel();
  
  if (weekendActivity) {
    console.log(`Weekend suggestion: ${weekendActivity.name}`);
    
    // Schedule for next Saturday
    const nextSaturday = new Date();
    nextSaturday.setDate(nextSaturday.getDate() + (6 - nextSaturday.getDay() + 7) % 7);
    
    scheduler.scheduleActivity(weekendActivity, nextSaturday, "First activity from our new system!");
  }
  
  // Check upcoming schedule
  const upcoming = scheduler.getUpcomingActivities();
  console.log(`You have ${upcoming.length} activities scheduled for the next week`);
  
  return {
    wheel: activityWheel,
    scheduler: scheduler
  };
}
